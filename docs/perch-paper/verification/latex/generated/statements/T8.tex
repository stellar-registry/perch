&\quad \land\ \mathit{Derives}(R, R')\\
&\quad \land\ \mathit{allows}(R', \mathsf{call}(C, f, \vec v), S)\\
&\qquad \implies C \neq A \;\land\; C \notin X\end{aligned}\]
